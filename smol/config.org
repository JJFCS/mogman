(defun onncera-smart-beginning-of-line ()
    "moves cursor to first non-whitespace char or beg of line - alternates"
    (interactive)
    (let ((old-point (point)))
        (back-to-indentation)
        (when (= old-point (point))
            (beginning-of-line)
        )
    )

(defun onncera/mark-word-forward  () "extend the region forward  by one word"
    (interactive) (unless (region-active-p) (set-mark (point))) (forward-word  1) (activate-mark))

(defun onncera/mark-word-backward () "extend the region backward by one word"
    (interactive) (unless (region-active-p) (set-mark (point))) (backward-word 1) (activate-mark))

;; @subtopic text selection
(defun onncera/mark-line-up ()   "extend the region upward   by one line"
    (interactive) (unless (region-active-p) (set-mark (point))) (forward-line -1) (activate-mark))

(defun onncera/mark-line-down () "extend the region downward by one line"
    (interactive) (unless (region-active-p) (set-mark (point))) (forward-line  1) (activate-mark))

#+end_src

(defun onncera/disable-italics ()
    (interactive)
    (dolist (face (face-list))
      (when (memq (face-attribute face :slant nil t)
            '(italic oblique))
            (set-face-attribute face nil :slant 'normal)
      )
    )
)
